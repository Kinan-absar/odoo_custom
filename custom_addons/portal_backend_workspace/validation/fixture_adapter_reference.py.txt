from pathlib import Path
import re, shutil, subprocess, json, ast, sys
B=Path('/mnt/data/pbw_parity_audit'); ROOT=B/'v091/portal_backend_workspace'; F=B/'fixtures091';F.mkdir(exist_ok=True)
mods={'sales':'absar_portal_sales','purchase':'absar_portal_purchase_monitor','accounting':'absar_portal_accounting','hr':'absar_portal_employee_admin'}
method_map={}
for p in (B/'original_main/absar_portal_purchase_monitor/models').glob('*.py'):
 target=ROOT/'models/purchase'/p.name
 if not target.exists():continue
 new={n.name for n in ast.walk(ast.parse(target.read_text())) if isinstance(n,ast.FunctionDef)}
 for n in ast.walk(ast.parse(p.read_text())):
  if isinstance(n,ast.FunctionDef) and n.name.replace('_portal_','_pbw_purchase_') in new and n.name.startswith('_portal_'):
   method_map[n.name]=n.name.replace('_portal_','_pbw_purchase_')
def adapt(t):
 for old,new in method_map.items():t=re.sub(r'\b'+re.escape(old)+r'\b',new,t)
 for part,mod in mods.items():
  t=t.replace(mod+'.','portal_backend_workspace.'+part+'__')
  t=t.replace('/'+mod+'/static/src/','/portal_backend_workspace/static/src/'+part+'/')
 for a,b in [('/my/employee/purchase-monitor','/my/workspace/purchase-monitor'),('/my/employee/admin','/my/workspace/hr'),('/my/employee/hr-admin','/my/workspace/hr'),('/my/employee','/my/workspace'),('portal.report.seen','pbw.notification.seen'),('construction_project_id','pbw_project_id'),('portal_accounting_','pbw_accounting_'),('absar.','pbw.'),('absar_','pbw_'),('absar-','pbw-'),('Absar','Pbw'),('ABSAR','PBW'),('aps-','pbw-sales-'),('apw-','pbw-purchase-'),('apm-','pbw-monitor-'),('aca-','pbw-accounting-'),('aea-','pbw-hr-'),('ep-','pbwep-'),('apsFormat','pbwSalesFormat'),('apsDiscard','pbwSalesDiscard'),('apsReadOnly','pbwSalesReadOnly'),('acaDiscard','pbwAccountingDiscard'),('aeaDiscard','pbwHrDiscard'),('aeaDirty','pbwHrDirty'),('aeaGenerateOpen','pbwHrGenerateOpen'),('aeaSuggest','pbwHrSuggest'),('aeaKind','pbwHrKind'),('aeaDiscard','pbwHrDiscard')]:t=t.replace(a,b)
 t=t.replace('pbw_portal_accounting/', 'accounting/')
 t=t.replace('pbw_portal_purchase_monitor/', 'purchase/')
 t=t.replace('pbw_portal_sales/', 'sales/')
 for part, mod in mods.items():
  for q in (B/('original_staging' if part=='hr' else 'original_main')/mod/'views').glob('*.xml'):
   from lxml import etree
   tree=etree.parse(str(q))
   for element in tree.xpath('//*[@id]'):
    old=element.get('id')
    if '.' not in old:
     t=t.replace('[@id="'+old+'"]', '[@id="'+part+'__'+old+'"]')
 t=re.sub(r'\baca(?=[\s\'".{}>,:()\[\]]|$)','pbw-accounting',t)
 t=re.sub(r'\baea(?=[\s\'".{}>,:()\[\]]|$)','pbw-hr',t)
 t=t.replace('Set the Bill Date before submitting for approval.', 'Set the invoice/bill date before submitting for approval.')
 return t
jobs=[]
for part,mod in mods.items():
 original=B/('original_staging' if part=='hr' else 'original_main')/mod
 target=F/part;target.mkdir(exist_ok=True)
 for d in ['models','views','controllers','security']:
  out=target/d
  if out.is_symlink():out.unlink()
  elif out.exists():shutil.rmtree(out)
  out.symlink_to(ROOT/d/part,target_is_directory=True)
 (target/'static').mkdir(exist_ok=True)
 out=target/'static/src'
 if out.is_symlink():out.unlink()
 elif out.exists():shutil.rmtree(out)
 out.symlink_to(ROOT/'static/src'/part,target_is_directory=True)
 folder='tools' if part in ('sales','purchase') else 'quality_checks' if part=='accounting' else 'tests'
 out=target/folder;out.mkdir(exist_ok=True)
 for p in sorted((original/folder).glob('*')):
  if p.suffix in ('.py','.mjs') and not p.name.startswith('__'):
   (out/p.name).write_text(adapt(p.read_text()))
 for p in sorted((original/folder).glob('*')):
  if p.suffix not in ('.py','.mjs') or p.name.startswith('__'):continue
  if p.name in ['odoo_runtime_cases.py','roles_cases.py','runtime_cases.py','design_cases.py']:continue
  q=out/p.name;q.write_text(adapt(p.read_text()))
  cmd=['node' if q.suffix=='.mjs' else sys.executable,str(q)]
  try:
   r=subprocess.run(cmd,text=True,capture_output=True,timeout=35)
   (out/(p.name+'.log')).write_text(r.stdout+r.stderr)
   jobs.append({'component':part,'check':p.name,'returncode':r.returncode,'output':(r.stdout+r.stderr)[-4500:]})
  except Exception as e:jobs.append({'component':part,'check':p.name,'returncode':-1,'output':str(e)})
  print(part,p.name,'PASS' if jobs[-1]['returncode']==0 else 'FAIL')
(F/'results.json').write_text(json.dumps(jobs,indent=2))
