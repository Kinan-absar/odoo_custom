def render(n,ctx,looped=False):
    if not isinstance(n.tag,str):return ''
    if 't-foreach' in n.attrib and not looped:
        return ''.join(render(n,dict(ctx,**{n.get('t-as'):item}),True) for item in eval(n.get('t-foreach'),{},ctx))
    if 't-if' in n.attrib and not eval(n.get('t-if'),{},ctx):return ''
    content=escape(str(eval(n.get('t-out'),{},ctx))) if 't-out' in n.attrib else (escape(n.text or '')+''.join(render(c,ctx)+escape(c.tail or '') for c in n))
    if n.tag in ('t','template'):return content
    attrs=[]
    for k,v in n.attrib.items():
        if k.startswith('t-att-'):
            val=eval(v,{},ctx)
            if val is None or val is False:continue
            attrs.append((k[6:],str(val)))
        elif not k.startswith('t-'):attrs.append((k,v))
    return '<'+n.tag+''.join(' %s="%s"'%(k,escape(v,quote=True)) for k,v in attrs)+'>'+content+'</'+n.tag+'>'

def mount(page,state,shell='workspace'):
    # Pure local HTML fixture. Production code is unchanged. The only JS adapter
    # supplies an HTTP origin for safeURL because about:blank has an opaque origin.
    import re
    text=re.sub(r'<link[^>]+>|<script[^>]+></script>', '', state.html(shell))
    page.set_content(text)
    page.add_style_tag(content=EPSCSS if shell=='eps' else CORE)
    page.add_style_tag(content=(ROOT/'static/src/core/notification_bell.css').read_text())
    page.expose_function('fixtureRequest',state.fixture_request)
    page.evaluate("""() => { window.fetch = async (url, options={}) => {
        const result = await window.fixtureRequest(String(url), options.method || 'GET', options.body || '');
        return new Response(JSON.stringify(result.data), {status:result.status, headers:{'Content-Type':'application/json'}});
    }; }""")
    code=(ROOT/'static/src/core/notification_bell.js').read_text()
    code=code.replace('location.origin', "'https://notification-fixture.invalid'")
    page.add_script_tag(content=code)
    page.wait_for_timeout(180)
