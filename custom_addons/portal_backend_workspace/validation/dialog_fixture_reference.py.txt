from pathlib import Path
import json
from playwright.sync_api import sync_playwright
B=Path('/mnt/data/pbw_parity_audit');OUT=B/'checks'
oldcss=(B/'original_main/absar_portal_accounting/static/src/accounting.css').read_text()
newcss=(B/'v091/portal_backend_workspace/static/src/accounting/accounting.css').read_text()
def document(css,root,prefix):
 return '''<!doctype html><html><head><meta charset="utf-8"><style>html,body{margin:0;padding:0}*{box-sizing:border-box}'''+css+'''</style></head><body><dialog open class="'''+root+' '+prefix+'''dialog"><h3>Product</h3><div class="'''+prefix+'''dialog-body"><label><span>Product Name</span><input value="Office supplies"/></label><label><span>Description</span><textarea>Order line description</textarea></label></div><div class="'''+prefix+'''dialog-actions"><button class="primary">Save</button><button>Cancel</button></div></dialog></body></html>'''
get='''() => Object.fromEntries(['dialog','label','input','textarea','button'].map(k=>{const e=document.querySelector(k),s=getComputedStyle(e),r=e.getBoundingClientRect();return [k,{fontFamily:s.fontFamily,fontSize:s.fontSize,padding:s.padding,border:s.border,borderRadius:s.borderRadius,color:s.color,backgroundColor:s.backgroundColor,boxSizing:s.boxSizing,width:r.width}]}))'''
results=[]
with sync_playwright() as p:
 browser=p.chromium.launch(headless=True,executable_path='/usr/bin/chromium',args=['--no-sandbox'])
 for width in [1280,390]:
  page=browser.new_page(viewport={'width':width,'height':850})
  page.route('**/*',lambda route:route.abort())
  samples={}
  for name,css,root,prefix in [('original',oldcss,'aca','aca-'),('v09',newcss,'aca','pbw-accounting-'),('v091',newcss,'pbw-accounting','pbw-accounting-')]:
   page.set_content(document(css,root,prefix));samples[name]=page.evaluate(get)
   if name=='v091':page.screenshot(path=str(OUT/f'accounting_dialog_{width}.png'),full_page=True)
  assert samples['original']==samples['v091'],json.dumps(samples,indent=2)
  assert samples['original']!=samples['v09'],'Fixture did not reproduce the missing root styles'
  results.append({'width':width,'original_vs_fixed':'MATCH','old_v09_regression_reproduced':True,'computed_styles':samples})
  page.close()
 browser.close()
(OUT/'dialog_style_results.json').write_text(json.dumps(results,indent=2))
print('PASS: original vs repaired Accounting dialog styles match at 1280px and 390px; v0.9 regression reproduced at both widths. Isolated real Chromium fixture, not an Odoo page.')
